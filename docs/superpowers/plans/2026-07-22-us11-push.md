# US-11 — Notifications push de rappel — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Envoyer une notification push à 17h30 heure locale à tout Héros dont le Boss du jour est encore vivant, via une nouvelle section « Intégrations » du QG (opt-in explicite) et un service worker déjà en place (`app/sw.ts`, Serwist).

**Architecture:** Table `push_subscriptions` (policy DELETE autorisée, hors RG-04) + colonne `profiles.last_reminder_sent_on` (idempotence). Sélection des profils éligibles = fonction pure testée (`lib/push-eligibility.ts`, 5 conditions du design) consommée par une route serverless (`app/api/cron/push-reminders/route.ts`, client Supabase service-role + `web-push`/VAPID). Le **palier Vercel Hobby limite les Cron Jobs natifs à 1 exécution/jour** (incompatible avec la fenêtre 17h30 multi-fuseaux qui a besoin d'un déclenchement ~15 min) : la route est donc appelée par un **workflow GitHub Actions planifié** plutôt que par `vercel.json`. Le contenu du message vient d'un module pur de gabarits (`lib/push-copy.ts`). Le SW existant (`app/sw.ts`, Serwist) reçoit deux listeners `self.addEventListener` (`push`, `notificationclick`) ajoutés juste après `serwist.addEventListeners()` — pas de second service worker. Souscription client via `registration.pushManager.subscribe()`, où `registration` vient d'un nouvel export de `lib/pwa/update-prompt-client.ts` (`getServiceWorkerRegistration`).

**Tech Stack:** Next.js 16 (App Router, route handler Node runtime), Supabase (`@supabase/supabase-js` service-role côté cron), `web-push` (VAPID), Serwist (`app/sw.ts` existant), Vitest, GitHub Actions (`schedule`).

```ts
// US-11 : contenu des notifications de rappel — ton "combat de rue" déjà
// établi ailleurs (combo, rage quit, Perfect/K.O.), pas une formule fixe.
// Module pur, même logique que lib/bosses.ts / lib/heroes.ts : pas mêlé à
// lib/game-engine.ts (réservé aux calculs RG-01/02).

export interface PushReminderPayload {
  title: string;
  body: string;
  url: string;
}

function pluralize(n: number, singular: string, plural: string): string {
  return n <= 1 ? singular : plural;
}

// Gabarits paramétrés par le nombre de tâches restantes. "Crépuscule" =
// renommage de l'ex-"Obi-Wan" (Rituel du Crépuscule, US-07).
const REMINDER_TEMPLATES: Array<(n: number) => string> = [
  (n) =>
    `Le Boss ricane, il te reste ${n} ${pluralize(n, "monstre", "monstres")} à abattre avant le Rituel du Crépuscule !`,
  (n) =>
    `Le Dojo t'attend : ${n} ${pluralize(n, "adversaire", "adversaires")} ${pluralize(n, "tient", "tiennent")} encore debout.`,
  (n) =>
    `Le Crépuscule approche et ton Boss respire encore (${n} ${pluralize(n, "tâche restante", "tâches restantes")}).`,
  (n) =>
    `Termine-le : ${n} ${pluralize(n, "point de vie", "points de vie")} planqué${n <= 1 ? "" : "s"} dans tes tâches du jour.`,
];

// Tire un gabarit (aléatoire par défaut, injectable pour les tests) et
// injecte le nombre de tâches restantes. `url: "/"` = deep-link vers
// l'Arène, interprété par le listener notificationclick du service worker.
export function buildReminderPayload(
  remainingTasks: number,
  pickIndex: (templateCount: number) => number = (count) => Math.floor(Math.random() * count),
): PushReminderPayload {
  const index = pickIndex(REMINDER_TEMPLATES.length);
  const template = REMINDER_TEMPLATES[index] ?? REMINDER_TEMPLATES[0];
  return {
    title: "Dodojo — Le Boss t'attend",
    body: template(remainingTasks),
    url: "/",
  };
}
```

```ts
import { calculateBossCurrentHp, dayKey, type HpTask } from "./game-engine";

export const REMINDER_HOUR = 17;
export const REMINDER_MINUTE = 30;
// Largeur totale de la fenêtre (±10 min autour de 17h30), calée sur la
// cadence du cron externe (~15 min, cf. Task 9) avec marge pour son jitter.
export const REMINDER_WINDOW_MINUTES = 20;

function localMinutesSinceMidnight(nowUtc: Date, timezone: string): number {
  const parts = new Intl.DateTimeFormat("en-GB", {
    timeZone: timezone,
    hour: "2-digit",
    minute: "2-digit",
    hourCycle: "h23",
  }).formatToParts(nowUtc);
  const hour = Number(parts.find((p) => p.type === "hour")?.value ?? "0");
  const minute = Number(parts.find((p) => p.type === "minute")?.value ?? "0");
  return hour * 60 + minute;
}

// Vrai si l'heure locale (fuseau IANA donné) tombe dans la fenêtre 17h30 ± 10 min.
export function isWithinReminderWindow(nowUtc: Date, timezone: string): boolean {
  const target = REMINDER_HOUR * 60 + REMINDER_MINUTE;
  const current = localMinutesSinceMidnight(nowUtc, timezone);
  return Math.abs(current - target) <= REMINDER_WINDOW_MINUTES / 2;
}

export interface ReminderCandidate {
  profileId: string;
  timezone: string;
  lastReminderSentOn: string | null;
  hasSubscription: boolean;
  todayTasks: HpTask[]; // tâches du jour LOCAL courant de ce profil
  dayResultExists: boolean; // ligne day_results pour (user_id, jour local courant)
}

// US-11 : les 5 conditions du design
// (docs/superpowers/specs/2026-07-22-us11-push-design.md).
export function isReminderEligible(candidate: ReminderCandidate, nowUtc: Date): boolean {
  if (!candidate.hasSubscription) return false; // (1) souscription
  if (!isWithinReminderWindow(nowUtc, candidate.timezone)) return false; // (2) fenêtre horaire
  const today = dayKey(nowUtc, candidate.timezone);
  if (candidate.lastReminderSentOn === today) return false; // (3) pas déjà envoyé aujourd'hui
  if (candidate.dayResultExists) return false; // (5) jour pas déjà clôturé
  return calculateBossCurrentHp(candidate.todayTasks) > 0; // (4) Boss encore vivant
}

export function selectReminderEligibleProfiles(
  candidates: ReminderCandidate[],
  nowUtc: Date,
): string[] {
  return candidates.filter((c) => isReminderEligible(c, nowUtc)).map((c) => c.profileId);
}
```

```ts
/// <reference lib="esnext" />
/// <reference lib="webworker" />
import { CacheFirst, ExpirationPlugin, Serwist } from "serwist";
import type { PrecacheEntry, SerwistGlobalConfig } from "serwist";
import type { PushReminderPayload } from "../lib/push-copy";

declare global {
  interface WorkerGlobalScope extends SerwistGlobalConfig {
    __SW_MANIFEST: (PrecacheEntry | string)[] | undefined;
  }
}

declare const self: ServiceWorkerGlobalScope;

// Chemins publics dont les images sont mises en cache runtime À LA DEMANDE
// (jamais précachées en bloc — 12 boss × dizaines de frames, l'utilisateur
// ne verra jamais la plupart des boss, cf.
// docs/superpowers/specs/2026-07-22-pwa-design.md) : sprites de boss/héros,
// décors, effets visuels.
const RUNTIME_CACHED_IMAGE_PREFIXES = ["/sprites/", "/backgrounds/", "/vfx/"];

const serwist = new Serwist({
  // self.__SW_MANIFEST est injecté par app/serwist/[path]/route.ts : les
  // chunks JS/CSS du build Next.js (hashés, donc déjà versionnés) +
  // additionalPrecacheEntries (police, manifest, page offline — voir route.ts).
  precacheEntries: self.__SW_MANIFEST,
  skipWaiting: true,
  clientsClaim: true,
  navigationPreload: true,
  runtimeCaching: [
    {
      matcher({ request }) {
        if (request.destination !== "image") return false;
        const { pathname } = new URL(request.url);
        return RUNTIME_CACHED_IMAGE_PREFIXES.some((prefix) => pathname.startsWith(prefix));
      },
      handler: new CacheFirst({
        cacheName: "dodojo-sprites",
        plugins: [
          new ExpirationPlugin({
            maxEntries: 250,
            maxAgeSeconds: 60 * 60 * 24 * 90, // 90 jours
          }),
        ],
      }),
    },
  ],
  fallbacks: {
    entries: [
      {
        url: "/offline",
        matcher({ request }) {
          return request.destination === "document";
        },
      },
    ],
  },
});

serwist.addEventListeners();

// US-11 : notification de rappel (payload JSON envoyé par
// app/api/cron/push-reminders/route.ts via web-push). "Plus tard" (action)
// referme sans ouvrir l'app — évite de forcer l'ouverture juste pour dismiss.
self.addEventListener("push", (event: PushEvent) => {
  if (!event.data) return;
  const payload = event.data.json() as PushReminderPayload;
  event.waitUntil(
    self.registration.showNotification(payload.title, {
      body: payload.body,
      icon: "/icons/icon-192.png",
      badge: "/icons/icon-192.png",
      data: { url: payload.url },
      actions: [{ action: "later", title: "Plus tard" }],
    }),
  );
});

// Clic sur la notification : focus un onglet existant sinon ouvre l'Arène
// (data.url). L'action "Plus tard" ferme déjà la notification (ligne
// suivante, commune à toutes les actions) sans rien ouvrir de plus.
self.addEventListener("notificationclick", (event: NotificationEvent) => {
  event.notification.close();
  if (event.action === "later") return;

  const url = (event.notification.data as { url?: string } | undefined)?.url ?? "/";
  event.waitUntil(
    self.clients.matchAll({ type: "window", includeUncontrolled: true }).then((clientsList) => {
      const existing = clientsList[0];
      if (existing) return existing.focus();
      return self.clients.openWindow(url);
    }),
  );
});
```

```ts
import { Serwist } from "@serwist/window";

// Singleton module : une seule registration par onglet, peu importe combien
// de composants montent usePwaUpdatePrompt.
let serwist: Serwist | null = null;
let updateAvailable = false;
const listeners = new Set<() => void>();

function emit(): void {
  for (const listener of listeners) listener();
}

// Enregistré uniquement en production : en dev, un SW qui cache des chunks
// JS interférerait avec le Fast Refresh de Next.js.
export function initUpdatePrompt(): void {
  if (serwist || typeof window === "undefined") return;
  if (!("serviceWorker" in navigator)) return;
  if (process.env.NODE_ENV !== "production") return;

  serwist = new Serwist("/serwist/sw.js", { scope: "/" });

  serwist.addEventListener("waiting", () => {
    updateAvailable = true;
    emit();
  });

  serwist.addEventListener("controlling", () => {
    window.location.reload();
  });

  void serwist.register();
}

export function subscribeUpdatePrompt(listener: () => void): () => void {
  listeners.add(listener);
  return () => listeners.delete(listener);
}

export function getUpdatePromptSnapshot(): boolean {
  return updateAvailable;
}

// Dit au SW en attente d'activer immédiatement (self.skipWaiting()) ; le
// listener "controlling" ci-dessus recharge la page une fois qu'il a pris
// le contrôle.
export function reloadForUpdate(): void {
  serwist?.messageSkipWaiting();
}

// US-11 : registration active du SW, pour souscrire au Web Push
// (registration.pushManager.subscribe). `navigator.serviceWorker.ready` se
// résout dès qu'un SW est actif pour ce scope, quelle que soit la façon dont
// il a été enregistré — évite de dépendre d'un détail interne de la classe
// Serwist. Timeout court : en dev (SW jamais enregistré, cf.
// initUpdatePrompt) ou navigateur non compatible, ne bloque jamais
// indéfiniment l'appelant.
export async function getServiceWorkerRegistration(): Promise<ServiceWorkerRegistration | null> {
  if (typeof window === "undefined" || !("serviceWorker" in navigator)) return null;

  const timeout = new Promise<null>((resolve) => setTimeout(() => resolve(null), 3000));
  const ready = navigator.serviceWorker.ready.catch(() => null);
  return Promise.race([ready, timeout]);
}
```

```ts
// Convertit la clé publique VAPID (base64url, NEXT_PUBLIC_VAPID_PUBLIC_KEY)
// en Uint8Array attendu par PushManager.subscribe({ applicationServerKey }).
// Algorithme standard (padding base64 + caractères url-safe → standards).
export function urlBase64ToUint8Array(base64String: string): Uint8Array {
  const padding = "=".repeat((4 - (base64String.length % 4)) % 4);
  const base64 = (base64String + padding).replace(/-/g, "+").replace(/_/g, "/");
  const rawData = atob(base64);
  const outputArray = new Uint8Array(rawData.length);
  for (let i = 0; i < rawData.length; i++) {
    outputArray[i] = rawData.charCodeAt(i);
  }
  return outputArray;
}
```

```ts
"use server";

import { createClient } from "@/lib/supabase/server";

export interface SerializedPushSubscription {
  endpoint: string;
  keys: { p256dh: string; auth: string };
}

interface PushActionResult {
  error: string | null;
}

// US-11 : enregistre une souscription push. Upsert par endpoint (unique en
// base) — une re-souscription après reload ne doit jamais dupliquer la ligne.
export async function subscribeToPush(
  subscription: SerializedPushSubscription,
): Promise<PushActionResult> {
  if (!subscription.endpoint || !subscription.keys?.p256dh || !subscription.keys?.auth) {
    return { error: "Souscription push invalide." };
  }

  const supabase = await createClient();
  const {
    data: { user },
  } = await supabase.auth.getUser();
  if (!user) return { error: "Session expirée. Reconnecte-toi." };

  const { error } = await supabase.from("push_subscriptions").upsert(
    {
      user_id: user.id,
      endpoint: subscription.endpoint,
      p256dh: subscription.keys.p256dh,
      auth: subscription.keys.auth,
    },
    { onConflict: "endpoint" },
  );

  if (error) return { error: "Impossible d'activer les rappels. Réessaie." };
  return { error: null };
}

// US-11 : désinscription explicite (toggle désactivé côté QG). Pas un DELETE
// soumis à RG-04 — donnée technique d'appareil, policy DELETE autorisée
// (cf. supabase/migrations/20260722090000_create_push_subscriptions.sql).
export async function unsubscribeFromPush(endpoint: string): Promise<PushActionResult> {
  const supabase = await createClient();
  const {
    data: { user },
  } = await supabase.auth.getUser();
  if (!user) return { error: "Session expirée. Reconnecte-toi." };

  const { error } = await supabase
    .from("push_subscriptions")
    .delete()
    .eq("user_id", user.id)
    .eq("endpoint", endpoint);

  if (error) return { error: "Impossible de désactiver les rappels. Réessaie." };
  return { error: null };
}
```

```tsx
"use client";

import { useEffect, useState, useTransition } from "react";
import { getServiceWorkerRegistration } from "@/lib/pwa/update-prompt-client";
import { urlBase64ToUint8Array } from "@/lib/pwa/push-subscribe";
import { subscribeToPush, unsubscribeFromPush } from "@/app/actions/push";

interface PushReminderToggleProps {
  hasSubscription: boolean; // état initial connu côté serveur (au chargement du QG)
}

type ToggleStatus = "unsupported" | "denied" | "off" | "on";

function readNotificationPermission(): NotificationPermission | "unsupported" {
  if (typeof window === "undefined" || !("Notification" in window)) return "unsupported";
  return Notification.permission;
}

function computeStatus(hasSubscription: boolean): ToggleStatus {
  const permission = readNotificationPermission();
  if (permission === "unsupported") return "unsupported";
  if (permission === "denied") return "denied";
  return hasSubscription && permission === "granted" ? "on" : "off";
}

// US-11 : toggle d'opt-in explicite — aucun prompt navigateur non sollicité
// (Notification.requestPermission() déclenché uniquement au clic ici).
// Reflète l'état RÉEL du navigateur (Notification.permission), pas
// seulement la ligne en base — cf. edge case "permission révoquée après
// souscription" du design.
export default function PushReminderToggle({ hasSubscription }: PushReminderToggleProps) {
  const [pending, startTransition] = useTransition();
  const [error, setError] = useState<string | null>(null);
  const [status, setStatus] = useState<ToggleStatus>(() => computeStatus(hasSubscription));

  useEffect(() => {
    setStatus(computeStatus(hasSubscription));
  }, [hasSubscription]);

  function activate() {
    setError(null);
    startTransition(async () => {
      const permission = await Notification.requestPermission();
      if (permission !== "granted") {
        setStatus(permission === "denied" ? "denied" : "off");
        return;
      }

      const registration = await getServiceWorkerRegistration();
      if (!registration) {
        setError("Rappels indisponibles (app non installée ou service worker inactif).");
        return;
      }

      const publicKey = process.env.NEXT_PUBLIC_VAPID_PUBLIC_KEY;
      if (!publicKey) {
        setError("Configuration push manquante.");
        return;
      }

      const pushSubscription = await registration.pushManager.subscribe({
        userVisibleOnly: true,
        applicationServerKey: urlBase64ToUint8Array(publicKey),
      });
      const json = pushSubscription.toJSON();
      if (!json.endpoint || !json.keys?.p256dh || !json.keys?.auth) {
        setError("Souscription push incomplète.");
        return;
      }

      const result = await subscribeToPush({
        endpoint: json.endpoint,
        keys: { p256dh: json.keys.p256dh, auth: json.keys.auth },
      });
      if (result.error) {
        setError(result.error);
        return;
      }
      setStatus("on");
    });
  }

  function deactivate() {
    setError(null);
    startTransition(async () => {
      const registration = await getServiceWorkerRegistration();
      const pushSubscription = await registration?.pushManager.getSubscription();
      const endpoint = pushSubscription?.endpoint;
      await pushSubscription?.unsubscribe();
      if (endpoint) await unsubscribeFromPush(endpoint);
      setStatus("off");
    });
  }

  if (status === "unsupported") {
    return (
      <p className="text-xs text-faint">🔔 Rappel du soir : indisponible sur ce navigateur/appareil.</p>
    );
  }

  return (
    <div className="flex flex-col gap-1">
      <div className="flex items-center justify-between gap-3">
        <p className="text-sm text-ink">🔔 Rappel du soir (17h30)</p>
        <button
          type="button"
          disabled={pending || status === "denied"}
          onClick={status === "on" ? deactivate : activate}
          className="rounded-md border border-border bg-bg px-3 py-1.5 font-display text-[10px] uppercase tracking-wide text-gold disabled:opacity-50"
        >
          {status === "on" ? "Désactiver" : status === "denied" ? "Bloqué par le navigateur" : "Activer"}
        </button>
      </div>
      {status === "denied" && (
        <p className="text-xs text-faint">
          Autorise les notifications dans les réglages du navigateur pour activer ce rappel.
        </p>
      )}
      {error && <p className="text-xs text-hp">{error}</p>}
    </div>
  );
}
```

```ts
// /day_results sans avoir à connaître le fuseau de chaque profil à l'avance.
function adjacentUtcDates(now: Date): string[] {
  const dayMs = 86_400_000;
  return [-1, 0, 1].map((offset) => new Date(now.getTime() + offset * dayMs).toISOString().slice(0, 10));
}

export async function GET(request: Request): Promise<Response> {
  const secret = process.env.CRON_SECRET;
  const authHeader = request.headers.get("authorization");
  if (!secret || authHeader !== `Bearer ${secret}`) {
    return NextResponse.json({ error: "unauthorized" }, { status: 401 });
  }

  const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const serviceKey = process.env.NEXT_PRIVATE_KEY_SUPABASE;
  const vapidPublicKey = process.env.NEXT_PUBLIC_VAPID_PUBLIC_KEY;
  const vapidPrivateKey = process.env.VAPID_PRIVATE_KEY;
  const vapidSubject = process.env.VAPID_SUBJECT;
  if (!supabaseUrl || !serviceKey || !vapidPublicKey || !vapidPrivateKey || !vapidSubject) {
    return NextResponse.json({ error: "misconfigured" }, { status: 500 });
  }
  webpush.setVapidDetails(vapidSubject, vapidPublicKey, vapidPrivateKey);

  const supabase = createClient(supabaseUrl, serviceKey, { auth: { persistSession: false } });
  const now = new Date();

  const { data: subscriptions } = await supabase
    .from("push_subscriptions")
    .select("user_id, endpoint, p256dh, auth");
  const subs = (subscriptions ?? []) as SubscriptionRow[];
  if (subs.length === 0) return NextResponse.json({ notified: 0 });

  const userIds = [...new Set(subs.map((s) => s.user_id))];
  const candidateDates = adjacentUtcDates(now);

  const [profilesResult, tasksResult, dayResultsResult] = await Promise.all([
    supabase.from("profiles").select("id, timezone, last_reminder_sent_on").in("id", userIds),
    supabase
      .from("tasks")
      .select("user_id, day_date, parent_id, is_abandoned, is_completed")
      .in("user_id", userIds)
      .in("day_date", candidateDates),
    supabase.from("day_results").select("user_id, day_date").in("user_id", userIds).in("day_date", candidateDates),
  ]);

  const profileById = new Map(((profilesResult.data ?? []) as ProfileRow[]).map((p) => [p.id, p]));
  const subsByUser = new Map<string, SubscriptionRow[]>();
  for (const sub of subs) {
    const bucket = subsByUser.get(sub.user_id) ?? [];
    bucket.push(sub);
    subsByUser.set(sub.user_id, bucket);
  }
  const tasksByUser = new Map<string, TaskRow[]>();
  for (const task of (tasksResult.data ?? []) as TaskRow[]) {
    const bucket = tasksByUser.get(task.user_id) ?? [];
    bucket.push(task);
    tasksByUser.set(task.user_id, bucket);
  }
  const dayResultKeys = new Set(
    ((dayResultsResult.data ?? []) as DayResultRow[]).map((r) => `${r.user_id}:${r.day_date}`),
  );

  const candidates: ReminderCandidate[] = userIds
    .map((id) => profileById.get(id))
    .filter((p): p is ProfileRow => Boolean(p))
    .map((profile) => {
      const today = dayKey(now, profile.timezone);
      const todayTasks = (tasksByUser.get(profile.id) ?? []).filter((t) => t.day_date === today);
      return {
        profileId: profile.id,
        timezone: profile.timezone,
        lastReminderSentOn: profile.last_reminder_sent_on,
        hasSubscription: (subsByUser.get(profile.id) ?? []).length > 0,
        todayTasks,
        dayResultExists: dayResultKeys.has(`${profile.id}:${today}`),
      };
    });

  const candidateByProfile = new Map(candidates.map((c) => [c.profileId, c]));
  const eligibleIds = selectReminderEligibleProfiles(candidates, now);
  let notified = 0;

  for (const profileId of eligibleIds) {
    const profile = profileById.get(profileId);
    const candidate = candidateByProfile.get(profileId);
    if (!profile || !candidate) continue;

    const today = dayKey(now, profile.timezone);
    const remaining = calculateBossCurrentHp(candidate.todayTasks);
    const payload = JSON.stringify(buildReminderPayload(remaining));

    let sentOnce = false;
    for (const sub of subsByUser.get(profileId) ?? []) {
      try {
        await webpush.sendNotification(
          { endpoint: sub.endpoint, keys: { p256dh: sub.p256dh, auth: sub.auth } },
          payload,
        );
        sentOnce = true;
      } catch (err) {
        const statusCode = (err as { statusCode?: number }).statusCode;
        if (statusCode === 404 || statusCode === 410) {
          await supabase.from("push_subscriptions").delete().eq("endpoint", sub.endpoint);
        }
      }
    }

    if (sentOnce) {
      await supabase.from("profiles").update({ last_reminder_sent_on: today }).eq("id", profileId);
      notified += 1;
    }
  }

  return NextResponse.json({ notified });
}
```

```ts
class FakeQueryBuilder implements PromiseLike<{ data: Row[]; error: null }> {
  private filters: Array<(row: Row) => boolean> = [];
  private mode: "select" | "update" | "delete" = "select";
  private patch: Row = {};

  constructor(private readonly rows: Row[]) {}

  select(): this {
    this.mode = "select";
    return this;
  }

  update(patch: Row): this {
    this.mode = "update";
    this.patch = patch;
    return this;
  }

  delete(): this {
    this.mode = "delete";
    return this;
  }

  eq(column: string, value: unknown): this {
    this.filters.push((row) => row[column] === value);
    return this;
  }

  in(column: string, values: unknown[]): this {
    this.filters.push((row) => values.includes(row[column]));
    return this;
  }

  private matchedIndexes(): number[] {
    return this.rows
      .map((row, index) => (this.filters.every((f) => f(row)) ? index : -1))
      .filter((index) => index !== -1);
  }

  private apply(): Row[] {
    const indexes = this.matchedIndexes();
    if (this.mode === "update") {
      for (const i of indexes) Object.assign(this.rows[i], this.patch);
    }
    if (this.mode === "delete") {
      for (const i of [...indexes].reverse()) this.rows.splice(i, 1);
    }
    return indexes.map((i) => this.rows[i]).filter((row): row is Row => Boolean(row));
  }

  then<TResult1 = { data: Row[]; error: null }, TResult2 = never>(
    onfulfilled?: ((value: { data: Row[]; error: null }) => TResult1 | PromiseLike<TResult1>) | null,
  ): PromiseLike<TResult1 | TResult2> {
    const data = this.apply();
    return Promise.resolve({ data, error: null as null }).then(onfulfilled ?? undefined) as PromiseLike<
      TResult1 | TResult2
    >;
  }
}
```
