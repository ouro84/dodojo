# US-03 — Intégration Siri (webhook iOS Shortcuts) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Permettre l'ajout d'une tâche à la voix via Siri (iOS Shortcuts) grâce à un webhook personnel authentifié par token, sans app native ni session Supabase.

**Architecture:** Un token secret par utilisateur, généré/régénéré depuis un nouveau bloc de la section « Intégrations » du QG (`WebhookTokenSection`, sibling de `PushReminderToggle`, US-11), stocké **hashé** (SHA-256, `profiles.webhook_token_hash`) — jamais en clair. Une route serverless `POST /api/webhook/add-task` (client Supabase **service-role**, sans cookie) vérifie le hash puis délègue l'insertion à `insertTaskForUser` (`lib/task-insertion.ts`), **extraite** du formulaire d'ajout existant (`addTask`, `app/actions/tasks.ts`) pour garantir un seul calcul RG-01/RG-02 (position, `day_date` local, HP du Boss) — réutilisable telle quelle par le futur US-22 (Gemini MCP). Validation de titre partagée (`lib/task-title.ts`, `maxLength` cohérent avec le formulaire).

**Tech Stack:** Next.js 16 (App Router, Route Handler `runtime = "nodejs"`), Supabase (`@supabase/supabase-js` service-role côté webhook), `node:crypto` (hash SHA-256 + comparaison temps constant), Vitest.

```ts
import { randomBytes, createHash, timingSafeEqual } from "node:crypto";

// 256 bits, hex-encodé → 64 caractères. Import relatif volontaire (node:crypto
// mis à part) : ce fichier est chargé directement par Vitest (voir Global
// Constraints, pas d'alias @/ dans les fichiers testés).
export const WEBHOOK_TOKEN_BYTES = 32;

// US-03 : token webhook personnel affiché en clair UNE seule fois côté QG
// (voir app/actions/webhook-token.ts) — jamais stocké tel quel, seul son hash
// SHA-256 est persisté (hashWebhookToken).
export function generateWebhookToken(): string {
  return randomBytes(WEBHOOK_TOKEN_BYTES).toString("hex");
}

// SHA-256 hex — suffisant pour un secret à haute entropie généré
// aléatoirement (≠ mot de passe utilisateur à faible entropie, qui
// nécessiterait bcrypt/argon2 avec sel).
export function hashWebhookToken(token: string): string {
  return createHash("sha256").update(token).digest("hex");
}

// Comparaison en temps constant : évite qu'un attaquant mesure la latence
// pour deviner le hash caractère par caractère. `storedHash` peut être
// absent (aucun token généré) ou malformé → toujours invalide, jamais de
// throw (timingSafeEqual lève si les deux Buffers n'ont pas la même
// longueur, d'où la vérification de longueur avant l'appel).
export function verifyWebhookToken(
  token: string,
  storedHash: string | null | undefined,
): boolean {
  if (!token || !storedHash) return false;
  const candidate = Buffer.from(hashWebhookToken(token), "hex");
  const stored = Buffer.from(storedHash, "hex");
  if (candidate.length !== stored.length) return false;
  return timingSafeEqual(candidate, stored);
}
```

// Contrainte partagée par le formulaire d'ajout in-app (AddTaskForm.tsx,
// SubtaskForm.tsx, maxLength=140, contrôle HTML uniquement) et toute entrée
// non-UI (US-03 webhook Siri, futur US-22 MCP Gemini) — une seule source de
// vérité pour la validation SERVEUR, jamais réinventée par appelant.
export const TASK_TITLE_MAX_LENGTH = 140;

export type TaskTitleValidation = { title: string } | { error: string };

export function validateTaskTitle(rawTitle: string): TaskTitleValidation {
  const title = rawTitle.trim();
  if (!title) {
    return { error: "Nomme ton monstre avant de l'invoquer." };
  }
  if (title.length > TASK_TITLE_MAX_LENGTH) {
    return { error: `Le nom du monstre est trop long (${TASK_TITLE_MAX_LENGTH} caractères max).` };
  }
  return { title };
}

// Fake Supabase minimal : couvre les deux chaînes utilisées par
// insertTaskForUser — select().eq().eq() (awaited directement, tableau) et
// insert().select().single() (une ligne).
class FakeQueryBuilder implements PromiseLike<{ data: Row[]; error: null }> {
  private filters: Array<(row: Row) => boolean> = [];
  private mode: "select" | "insert" = "select";
  private insertedRow: Row | null = null;

  constructor(private readonly rows: Row[]) {}

  select(): this {
    return this;
  }

  insert(row: Row): this {
    this.mode = "insert";
    this.insertedRow = {
      id: `task-${this.rows.length + 1}`,
      parent_id: null,
      is_abandoned: false,
      is_completed: false,
      ...row,
    };
    return this;
  }

  eq(column: string, value: unknown): this {
    this.filters.push((row) => row[column] === value);
    return this;
  }

  private matched(): Row[] {
    return this.rows.filter((row) => this.filters.every((f) => f(row)));
  }

  async single(): Promise<{ data: Row | null; error: null }> {
    if (this.mode === "insert" && this.insertedRow) {
      this.rows.push(this.insertedRow);
      return { data: this.insertedRow, error: null };
    }
    return { data: this.matched()[0] ?? null, error: null };
  }

  then<TResult1 = { data: Row[]; error: null }, TResult2 = never>(
    onfulfilled?: ((value: { data: Row[]; error: null }) => TResult1 | PromiseLike<TResult1>) | null,
  ): PromiseLike<TResult1 | TResult2> {
    const data = this.matched();
    return Promise.resolve({ data, error: null as null }).then(onfulfilled ?? undefined) as PromiseLike<
      TResult1 | TResult2
    >;
  }
}

import type { SupabaseClient } from "@supabase/supabase-js";
import { calculateBossCurrentHp, calculateBossMaxHp, dayKey, nextSiblingPosition, type HpTask } from "./game-engine";

export interface InsertTaskParams {
  userId: string;
  title: string;
  timezone: string;
}

export interface InsertTaskOutcome {
  taskId: string;
  dayDate: string;
  bossMaxHp: number;
  bossCurrentHp: number;
}

type DayTaskRow = HpTask & { position: number };

// US-03/US-22 : logique d'insertion de tâche + recalcul des HP du Boss
// (RG-01/RG-02), extraite du formulaire d'ajout in-app (app/actions/tasks.ts,
// addTask) pour être réutilisée par une route API SANS session cookie —
// paramétrée par un user_id EXPLICITE (jamais supabase.auth.getUser() ici).
// Le webhook Siri (US-03) et le futur serveur MCP Gemini (US-22) l'appellent
// avec un client Supabase service-role ; addTask l'appelle avec le client
// cookie-session habituel. `title` est supposé déjà validé/trim() par
// l'appelant (voir lib/task-title.ts) — pas de re-validation ici. Import
// relatif volontaire (pas d'alias @/) : ce fichier est chargé directement
// par Vitest (voir Global Constraints du plan US-03).
export async function insertTaskForUser(
  supabase: SupabaseClient,
  { userId, title, timezone }: InsertTaskParams,
): Promise<InsertTaskOutcome> {
  const dayDate = dayKey(new Date(), timezone);

  const { data: todayTasks, error: fetchError } = await supabase
    .from("tasks")
    .select("parent_id, is_abandoned, is_completed, position")
    .eq("user_id", userId)
    .eq("day_date", dayDate);

  if (fetchError) {
    throw new Error("insert_task_fetch_failed");
  }

  const rows = (todayTasks ?? []) as DayTaskRow[];
  const roots = rows.filter((task) => task.parent_id === null);

  const { data: inserted, error: insertError } = await supabase
    .from("tasks")
    .insert({
      user_id: userId,
      title,
      day_date: dayDate,
      position: nextSiblingPosition(roots),
    })
    .select("id")
    .single();

  if (insertError || !inserted) {
    throw new Error("insert_task_failed");
  }

  const tasksAfterInsert: HpTask[] = [
    ...rows,
    { parent_id: null, is_abandoned: false, is_completed: false },
  ];

  return {

class FakeQueryBuilder implements PromiseLike<{ data: Row[]; error: null }> {
  private filters: Array<(row: Row) => boolean> = [];
  private mode: "select" | "insert" | "update" = "select";
  private patch: Row = {};
  private insertedRow: Row | null = null;

  constructor(private readonly rows: Row[]) {}

  select(): this {
    return this;
  }

  insert(row: Row): this {
    this.mode = "insert";
    this.insertedRow = {
      id: `task-${this.rows.length + 1}`,
      parent_id: null,
      is_abandoned: false,
      is_completed: false,
      ...row,
    };
    return this;
  }

  update(patch: Row): this {
    this.mode = "update";
    this.patch = patch;
    return this;
  }

  eq(column: string, value: unknown): this {
    this.filters.push((row) => row[column] === value);
    return this;
  }

  private matched(): Row[] {
    return this.rows.filter((row) => this.filters.every((f) => f(row)));
  }

  private apply(): Row[] {
    if (this.mode === "insert" && this.insertedRow) {
      this.rows.push(this.insertedRow);
      return [this.insertedRow];
    }
    if (this.mode === "update") {
      const matched = this.matched();
      for (const row of matched) Object.assign(row, this.patch);
      return matched;
    }
    return this.matched();
  }

  async maybeSingle(): Promise<{ data: Row | null; error: null }> {
    const rows = this.apply();
    return { data: rows[0] ?? null, error: null };
  }

  async single(): Promise<{ data: Row | null; error: null }> {
    const rows = this.apply();
    return { data: rows[0] ?? null, error: null };
  }

  then<TResult1 = { data: Row[]; error: null }, TResult2 = never>(
    onfulfilled?: ((value: { data: Row[]; error: null }) => TResult1 | PromiseLike<TResult1>) | null,
  ): PromiseLike<TResult1 | TResult2> {
    const data = this.apply();
    return Promise.resolve({ data, error: null as null }).then(onfulfilled ?? undefined) as PromiseLike<
      TResult1 | TResult2
    >;
  }
}

import { validateTaskTitle } from "../../../../lib/task-title";

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

interface WebhookProfileRow {
  id: string;
  timezone: string;
}

function extractBearerToken(header: string | null): string | null {
  if (!header) return null;
  const [scheme, value] = header.split(" ");
  if (scheme !== "Bearer" || !value) return null;
  return value;
}

// action "Obtenir le contenu de l'URL"). Auth par token personnel hashé
// (profiles.webhook_token_hash) — jamais de cookie, jamais d'OAuth (pas de
// flow interactif possible côté Shortcuts). Réutilise insertTaskForUser
// (lib/task-insertion.ts) : même logique d'insertion + recalcul HP que le
// formulaire d'ajout in-app, pas de duplication (RG-01/RG-02). Client
// Supabase service-role (comme app/api/cron/push-reminders, US-11) — hors
// RLS, user_id retrouvé explicitement via le hash du token.
export async function POST(request: Request): Promise<Response> {
  const token = extractBearerToken(request.headers.get("authorization"));
  if (!token) {
    return NextResponse.json({ error: "unauthorized" }, { status: 401 });
  }

  const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const serviceKey = process.env.NEXT_PRIVATE_KEY_SUPABASE;
  if (!supabaseUrl || !serviceKey) {
    return NextResponse.json({ error: "misconfigured" }, { status: 500 });
  }
  const supabase = createClient(supabaseUrl, serviceKey, { auth: { persistSession: false } });

  const { data: profile } = await supabase
    .from("profiles")
    .select("id, timezone")
    .eq("webhook_token_hash", hashWebhookToken(token))
    .maybeSingle();

  if (!profile) {
    return NextResponse.json({ error: "unauthorized" }, { status: 401 });
  }
  const profileRow = profile as WebhookProfileRow;

  let body: unknown;
  try {
    body = await request.json();
  } catch {
    return NextResponse.json({ error: "invalid_body" }, { status: 400 });
  }
  const rawTitle = (body as { title?: unknown } | null)?.title;
  if (typeof rawTitle !== "string") {
    return NextResponse.json({ error: "invalid_body" }, { status: 400 });
  }

  const validation = validateTaskTitle(rawTitle);
  if ("error" in validation) {
    return NextResponse.json({ error: validation.error }, { status: 400 });
  }

  const outcome = await insertTaskForUser(supabase, {
    userId: profileRow.id,
    title: validation.title,
    timezone: profileRow.timezone,
  });

  await supabase
    .from("profiles")
    .update({ webhook_token_last_used_at: new Date().toISOString() })
    .eq("id", profileRow.id);

  return NextResponse.json(
    {
      id: outcome.taskId,
      dayDate: outcome.dayDate,
      bossMaxHp: outcome.bossMaxHp,
      bossCurrentHp: outcome.bossCurrentHp,
    },

import { createClient } from "@/lib/supabase/server";
import { generateWebhookToken, hashWebhookToken } from "@/lib/webhook-token";

export interface WebhookTokenActionResult {
  error: string | null;
  token?: string;
  createdAt?: string;
}

// US-03 : (ré)génère le token webhook personnel (Siri/iOS Shortcuts). Écrase
// l'ancien hash — invalidation immédiate, sans DELETE (pas une donnée RG-04,
// pas un historique de combat). Le token en clair n'est renvoyé qu'ICI, une
// seule fois : jamais relu depuis la base ensuite (seul son hash SHA-256 est
// persisté, cf. lib/webhook-token.ts). webhook_token_last_used_at repart à
// null : la précédente "dernière utilisation" concernait l'ancien token.
export async function regenerateWebhookToken(): Promise<WebhookTokenActionResult> {
  const supabase = await createClient();
  const {
    data: { user },
  } = await supabase.auth.getUser();
  if (!user) return { error: "Session expirée. Reconnecte-toi." };

  const token = generateWebhookToken();
  const createdAt = new Date().toISOString();

  const { error } = await supabase
    .from("profiles")
    .update({
      webhook_token_hash: hashWebhookToken(token),
      webhook_token_created_at: createdAt,
      webhook_token_last_used_at: null,
    })
    .eq("id", user.id);

  if (error) return { error: "Impossible de générer le token. Réessaie." };
  return { error: null, token, createdAt };
}
