# US-09 : Progression de base (XP & Niveaux) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Faire gagner de l'XP au Héros à chaque tâche validée, le faire monter de niveau selon un seuil croissant, et afficher un écran "LEVEL UP" — tout en gérant symétriquement le décochage (retrait d'XP, possible delevel).

**Architecture:** Une fonction pure `applyXpDelta` dans `lib/game-engine.ts` porte toute la mécanique de seuils (montée et descente symétriques). Le server action `setTaskCompleted` l'utilise pour persister `profiles.level/xp` uniquement sur transition réelle. Le client (`Arena.tsx`) applique le même calcul de façon optimiste (même patron que le calcul HP existant), affiche un floating text XP indépendant du floating text HP, et un overlay `LevelUpOverlay` calqué sur `PerfectOverlay` existant.

**Tech Stack:** Next.js 16 App Router, TypeScript strict, Supabase (PostgreSQL), Tailwind CSS v4, Vitest.

## Global Constraints

- TypeScript strict : jamais de `any` (règle du projet, `CLAUDE.md`).
- Logique de jeu (calcul XP/niveau) exclusivement dans `lib/game-engine.ts`, testable sans dépendance UI (`CLAUDE.md` règle 2).
- Commits format `type: description courte` (Conventional Commits) — ici uniquement `feat:` et éventuellement `test:`/`refactor:` si un commit isole un refacto pur.
- Aucun commit direct sur `main` — ce plan s'exécute sur la branche `docs/us-09-xp-levels-design` déjà créée (ou une branche dédiée `feat/us-09-xp-levels` si on préfère séparer spec et code).
- Communication et tout texte utilisateur en français.
- XP par tâche principale validée : **30 XP fixe** (`XP_PER_TASK`).
- Seuil de niveau croissant : passer du niveau N à N+1 coûte **N × 100 XP** (`XP_THRESHOLD_PER_LEVEL = 100`, seuil = `xpThresholdForLevel(level)`).
- Décocher une tâche retire l'XP symétriquement, avec delevel en cascade si nécessaire ; plancher niveau 1 / XP 0 (jamais négatif).
- `profiles.current_title` reste inchangé (`'Novice'`) — hors scope explicite de ce lot.
- Zéro PR poussée tant que `npm test` et le build n'ont pas été vérifiés en local (`CLAUDE.md`).

```ts
export const XP_PER_TASK = 30;
export const XP_THRESHOLD_PER_LEVEL = 100;

export function xpThresholdForLevel(level: number): number {
  return level * XP_THRESHOLD_PER_LEVEL;
}

export interface XpState {
  level: number;
  xp: number;
}

export function applyXpDelta(state: XpState, delta: number): XpState {
  let level = state.level;
  let xp = state.xp + delta;

  while (xp >= xpThresholdForLevel(level)) {
    xp -= xpThresholdForLevel(level);
    level += 1;
  }

  while (xp < 0) {
    if (level <= 1) {
      level = 1;
      xp = 0;
      break;
    }
    level -= 1;
    xp += xpThresholdForLevel(level);
  }

  return { level, xp };
}
```

```ts
export async function setTaskCompleted(
  id: string,
  isCompleted: boolean,
): Promise<TaskActionResult> {
  const supabase = await createClient();
  const {
    data: { user },
  } = await supabase.auth.getUser();

  if (!user) {
    return { error: "Session expirée. Reconnecte-toi." };
  }

  const { data: existingTask, error: fetchError } = await supabase
    .from("tasks")
    .select("is_completed, is_abandoned")
    .eq("id", id)
    .eq("user_id", user.id)
    .maybeSingle();

  if (fetchError || !existingTask) {
    return {
      error: isCompleted
        ? "Le coup n'est pas passé. Réessaie."
        : "Impossible d'annuler le coup. Réessaie.",
    };
  }

  const { error: updateError } = await supabase
    .from("tasks")
    .update({ is_completed: isCompleted })
    .eq("id", id)
    .eq("user_id", user.id);

  if (updateError) {
    return {
      error: isCompleted
        ? "Le coup n'est pas passé. Réessaie."
        : "Impossible d'annuler le coup. Réessaie.",
    };
  }

  const isRealTransition = existingTask.is_completed !== isCompleted;
  let level: number | undefined;
  let xp: number | undefined;
  let leveledUp = false;

  if (isRealTransition && !existingTask.is_abandoned) {
    const { data: profile, error: profileError } = await supabase
      .from("profiles")
      .select("level, xp")
      .eq("id", user.id)
      .maybeSingle();

    // Le profil n'est jamais absent en usage normal (onboarding le garantit) ;
    // si le fetch échoue quand même, on ne bloque pas la validation de la
    // tâche pour autant — seul le gain d'XP est perdu pour ce coup.
    if (!profileError && profile) {
      const delta = isCompleted ? XP_PER_TASK : -XP_PER_TASK;
      const next = applyXpDelta(profile, delta);
      leveledUp = next.level > profile.level;
      level = next.level;
      xp = next.xp;

      await supabase
        .from("profiles")
        .update({ level: next.level, xp: next.xp })
        .eq("id", user.id);
    }
  }

  revalidatePath("/");
  return { error: null, level, xp, leveledUp };
}
```

```tsx
"use client";

import { useEffect, useRef, useState } from "react";
import type { Profile, Task } from "@/types";
import {
  applyXpDelta,
  calculateBossCurrentHp,
  calculateBossMaxHp,
  calculateHeroHp,
  HERO_MAX_HP,
  XP_PER_TASK,
} from "@/lib/game-engine";
import {
  addTask as addTaskAction,
  abandonTask as abandonTaskAction,
  setTaskCompleted as setTaskCompletedAction,
} from "@/app/actions/tasks";
import BottomNav from "@/components/ui/BottomNav";
import VersusScene from "./VersusScene";
import TaskList from "./TaskList";
import AddTaskForm from "./AddTaskForm";
import FloatingText from "./FloatingText";
import PerfectOverlay from "./PerfectOverlay";
import LevelUpOverlay from "./LevelUpOverlay";
import ImpactVfx, { pickImpactVariant, type ImpactVariantId } from "./ImpactVfx";

const BOSS_NAME = "Le Gardien de Glace";
const HERO_NAME = "Le Chevalier de Feu";

interface ArenaProps {
  initialProfile: Profile;
  initialTasks: Task[];
  overdueTasks: Task[];
}

interface Pulse {
  key: number;
  amount: number;
  kind: "damage" | "heal";
  impactVariant?: ImpactVariantId;
}

interface XpPulse {
  key: number;
  amount: number;
  kind: "damage" | "heal";
}

let pulseCounter = 0;
let xpPulseCounter = 0;
let optimisticIdCounter = 0;

function todayIso(): string {
  return new Date().toISOString().slice(0, 10);
}

export default function Arena({ initialProfile, initialTasks, overdueTasks }: ArenaProps) {
  const [profile, setProfile] = useState<Profile>(initialProfile);
  const [tasks, setTasks] = useState<Task[]>(initialTasks);
  const [pulse, setPulse] = useState<Pulse | null>(null);
  const [xpPulse, setXpPulse] = useState<XpPulse | null>(null);
  const [showPerfect, setShowPerfect] = useState(false);
  const [levelUpValue, setLevelUpValue] = useState<number | null>(null);
  const [banner, setBanner] = useState<string | null>(null);
  const [isAdding, setIsAdding] = useState(false);
  const pulseTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
  const xpPulseTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);

  const maxHp = calculateBossMaxHp(tasks);
  const currentHp = calculateBossCurrentHp(tasks);
  const heroHp = calculateHeroHp(overdueTasks);
  const activeTasks = tasks.filter((task) => !task.is_abandoned);
  const isEmpty = activeTasks.length === 0;

  useEffect(() => {
    return () => {
      if (pulseTimeoutRef.current) clearTimeout(pulseTimeoutRef.current);
      if (xpPulseTimeoutRef.current) clearTimeout(xpPulseTimeoutRef.current);
    };
  }, []);

  function triggerPulse(amount: number, kind: Pulse["kind"]) {
    pulseCounter += 1;
    const key = pulseCounter;
    setPulse({
      key,
      amount,
      kind,
      impactVariant: kind === "damage" ? pickImpactVariant() : undefined,
    });

    if (pulseTimeoutRef.current) clearTimeout(pulseTimeoutRef.current);
    pulseTimeoutRef.current = setTimeout(() => {
      setPulse((current) => (current?.key === key ? null : current));
    }, 1400);
  }

  function triggerXpPulse(amount: number, kind: XpPulse["kind"]) {
    xpPulseCounter += 1;
    const key = xpPulseCounter;
    setXpPulse({ key, amount, kind });

    if (xpPulseTimeoutRef.current) clearTimeout(xpPulseTimeoutRef.current);
    xpPulseTimeoutRef.current = setTimeout(() => {
      setXpPulse((current) => (current?.key === key ? null : current));
    }, 1400);
  }

  function handleAdd(title: string) {
    optimisticIdCounter += 1;
    const optimisticTask: Task = {
      id: `optimistic-${optimisticIdCounter}`,
      user_id: profile.id,
      parent_id: null,
      title,
      is_completed: false,
      is_abandoned: false,
      is_recurring: false,
      deadline_at: null,
      day_date: todayIso(),
      position: tasks.length,
      created_at: new Date().toISOString(),
    };

    setTasks((prev) => [...prev, optimisticTask]);
    triggerPulse(20, "heal");
    setIsAdding(true);

    void addTaskAction(title).then((result) => {
      setIsAdding(false);
      if (result.error) {
        setTasks((prev) => prev.filter((task) => task.id !== optimisticTask.id));
        setBanner(result.error);
      }
    });
  }

  function handleToggle(id: string, isCompleted: boolean) {
    const target = tasks.find((task) => task.id === id);
    if (!target || target.is_completed === isCompleted) return;

    const previousTasksSnapshot = tasks;
    const previousProfileSnapshot = profile;
    const previousHp = currentHp;
    const nextTasks = tasks.map((task) =>
      task.id === id ? { ...task, is_completed: isCompleted } : task,
    );

    setTasks(nextTasks);

    const nextHp = calculateBossCurrentHp(nextTasks);
    const delta = previousHp - nextHp;

    if (isCompleted) {
      triggerPulse(delta, "damage");
      if (nextHp === 0 && calculateBossMaxHp(nextTasks) > 0) {
        setShowPerfect(true);
      }
    } else {
      triggerPulse(-delta, "heal");
    }

    if (!target.is_abandoned) {
      const xpDelta = isCompleted ? XP_PER_TASK : -XP_PER_TASK;
      const nextProfile = applyXpDelta(profile, xpDelta);
      const didLevelUp = nextProfile.level > profile.level;

      setProfile((current) => ({ ...current, level: nextProfile.level, xp: nextProfile.xp }));
      triggerXpPulse(XP_PER_TASK, isCompleted ? "heal" : "damage");
      if (didLevelUp) {
        setLevelUpValue(nextProfile.level);
      }
    }

    void setTaskCompletedAction(id, isCompleted).then((result) => {
      if (result.error) {
        setTasks(previousTasksSnapshot);
        setProfile(previousProfileSnapshot);
        setLevelUpValue(null);
        if (isCompleted) setShowPerfect(false);
        setBanner(result.error);
        return;
      }

      const { level, xp } = result;
      if (level !== undefined && xp !== undefined) {
        setProfile((current) => ({ ...current, level, xp }));
      }
    });
  }

  function handleAbandon(id: string) {
    const previousSnapshot = tasks;
    const nextTasks = tasks.map((task) =>
      task.id === id ? { ...task, is_abandoned: true } : task,
    );

    setTasks(nextTasks);

    void abandonTaskAction(id).then((result) => {
      if (result.error) {
        setTasks(previousSnapshot);
        setBanner(result.error);
      }
    });
  }

  return (
    <div className="flex flex-1 flex-col gap-3 px-4 py-4">
      <div
        key={pulse ? `fx-${pulse.key}` : "fx-idle"}
        className={`relative flex flex-col gap-3 ${
          pulse?.kind === "damage" ? "is-hit" : ""
        }`}
      >
        {pulse?.kind === "damage" && (
          <div className="hit-flash-overlay pointer-events-none absolute inset-0 z-30 rounded-lg" />
        )}
        <div className="relative">
          <VersusScene
            heroName={HERO_NAME}
            heroHp={heroHp}
            heroMaxHp={HERO_MAX_HP}
            bossName={BOSS_NAME}
            currentHp={currentHp}
            maxHp={maxHp}
            dimmed={isEmpty}
          />
          {pulse?.kind === "damage" && pulse.impactVariant && (
            <ImpactVfx key={pulse.key} variant={pulse.impactVariant} />
          )}
          {pulse && <FloatingText amount={pulse.amount} kind={pulse.kind} />}
          {xpPulse && (
            <FloatingText amount={xpPulse.amount} kind={xpPulse.kind} unit="XP" />
          )}
          {showPerfect && (
            <PerfectOverlay
              bossName={BOSS_NAME}
              onDismiss={() => setShowPerfect(false)}
            />
          )}
          {levelUpValue !== null && (
            <LevelUpOverlay
              newLevel={levelUpValue}
              onDismiss={() => setLevelUpValue(null)}
            />
          )}
        </div>
      </div>

      {banner && (
        <div
          role="alert"
          className="flex items-center justify-between gap-2 rounded-md border border-alert bg-surface px-3 py-2 text-xs text-alert"
        >
          <span>{banner}</span>
          <button
            type="button"
            onClick={() => setBanner(null)}
            className="shrink-0 underline"
          >
            Fermer
          </button>
        </div>
      )}

      {isEmpty ? (
        <div className="flex flex-1 flex-col items-center justify-center gap-2 text-center">
          <p className="font-display text-lg uppercase tracking-wide text-ink">
            Terrain vague.
          </p>
          <p className="max-w-[24ch] text-sm text-muted">
            Aucun monstre à l&apos;horizon. La rue est calme… trop calme.
          </p>
        </div>
      ) : (
        <TaskList
          tasks={activeTasks}
          onToggle={handleToggle}
          onAbandon={handleAbandon}
        />
      )}

      <AddTaskForm onAdd={handleAdd} pending={isAdding} />

      <BottomNav active="arene" />
    </div>
  );
```
