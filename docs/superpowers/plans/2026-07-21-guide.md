```ts
export type GuideBlock =
  | { type: "h"; text: string }
  | { type: "p"; text: string }
  | { type: "ul"; items: string[] };

export interface GuideSection {
  id: string;
  title: string;
  blocks: GuideBlock[];
}

export interface Guide {
  updatedAt: string;
  sections: GuideSection[];
}

export interface NotionBlockLike {
  type: string;
  [key: string]: unknown;
}

function plain(node: unknown): string {
  const rt = (node as { rich_text?: Array<{ plain_text?: string }> })?.rich_text;
  if (!Array.isArray(rt)) return "";
  return rt.map((r) => r.plain_text ?? "").join("").trim();
}

function slug(title: string, i: number): string {
  const base = title
    .toLowerCase()
    .normalize("NFD")
    .replace(/[̀-ͯ]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");
  return base || `section-${i}`;
}

// Mappe des blocs Notion (plats) vers des sections internes. Un heading ouvre
// une nouvelle section ; paragraphes/listes se rattachent à la section courante.
// Les puces consécutives sont regroupées dans un seul bloc `ul`. Types non gérés
// et blocs vides ignorés.
export function mapNotionBlocks(blocks: NotionBlockLike[]): GuideSection[] {
  const sections: GuideSection[] = [];
  let current: GuideSection | null = null;

  const ensure = () => {
    if (!current) {
      current = { id: slug("", sections.length), title: "", blocks: [] };
      sections.push(current);
    }
    return current;
  };

  for (const block of blocks) {
    const t = block.type;
    if (t === "heading_1" || t === "heading_2" || t === "heading_3") {
      const title = plain(block[t]);
      current = { id: slug(title, sections.length), title, blocks: [] };
      sections.push(current);
      continue;
    }
    if (t === "paragraph") {
      const text = plain(block.paragraph);
      if (text) ensure().blocks.push({ type: "p", text });
      continue;
    }
    if (t === "bulleted_list_item" || t === "numbered_list_item") {
      const item = plain(block[t]);
      if (!item) continue;
      const sec = ensure();
      const last = sec.blocks[sec.blocks.length - 1];
      if (last && last.type === "ul") last.items.push(item);
      else sec.blocks.push({ type: "ul", items: [item] });
      continue;
    }
    // autres types ignorés
  }
  return sections;
}
```

```js
// Sync Notion → content/guide.json. Hors build prod. Usage: npm run sync:guide
// Requiert NOTION_TOKEN + NOTION_GUIDE_PAGE_ID. Sans eux : message clair, pas d'écrasement.
import { promises as fs } from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";

const ROOT = path.resolve(path.dirname(fileURLToPath(import.meta.url)), "..");
const OUT = path.join(ROOT, "content", "guide.json");

const token = process.env.NOTION_TOKEN;
const pageId = process.env.NOTION_GUIDE_PAGE_ID;
if (!token || !pageId) {
  console.error("NOTION_TOKEN / NOTION_GUIDE_PAGE_ID manquants — sync ignorée, guide.json conservé.");
  process.exit(0);
}

// A2 : import tolérant — un clone sans devDeps ne plante pas.
let Client;
try {
  ({ Client } = await import("@notionhq/client"));
} catch {
  console.error("@notionhq/client absent — lance `npm i -D @notionhq/client` pour activer la sync. guide.json conservé.");
  process.exit(0);
}

const notion = new Client({ auth: token });
const plain = (node) => (node?.rich_text ?? []).map((r) => r.plain_text ?? "").join("").trim();
const slug = (t, i) => (t.toLowerCase().normalize("NFD").replace(/[̀-ͯ]/g, "").replace(/[^a-z0-9]+/g, "-").replace(/^-+|-+$/g, "")) || `section-${i}`;

async function allBlocks(id) {
  const out = [];
  let cursor;
  do {
    const res = await notion.blocks.children.list({ block_id: id, start_cursor: cursor });
    out.push(...res.results);
    cursor = res.has_more ? res.next_cursor : undefined;
  } while (cursor);
  return out;
}

function mapBlocks(blocks) {
  const sections = [];
  let cur = null;
  const ensure = () => (cur ??= (sections.push((cur = { id: slug("", sections.length), title: "", blocks: [] })), cur));
  for (const b of blocks) {
    const t = b.type;
    if (t === "heading_1" || t === "heading_2" || t === "heading_3") {
      const title = plain(b[t]); cur = { id: slug(title, sections.length), title, blocks: [] }; sections.push(cur); continue;
    }
    if (t === "paragraph") { const text = plain(b.paragraph); if (text) ensure().blocks.push({ type: "p", text }); continue; }
    if (t === "bulleted_list_item" || t === "numbered_list_item") {
      const item = plain(b[t]); if (!item) continue;
      const s = ensure(); const last = s.blocks[s.blocks.length - 1];
      if (last?.type === "ul") last.items.push(item); else s.blocks.push({ type: "ul", items: [item] });
    }
  }
  return sections;
}

const sections = mapBlocks(await allBlocks(pageId));
const today = new Date().toISOString().slice(0, 10);
await fs.writeFile(OUT, JSON.stringify({ updatedAt: today, sections }, null, 2) + "\n");
console.log(`guide.json écrit : ${sections.length} sections.`);
```

```tsx
import type { GuideSection } from "@/lib/guide";

export default function GuideSections({ sections }: { sections: GuideSection[] }) {
  return (
    <div className="flex flex-col gap-6">
      {sections.map((s) => (
        <section key={s.id} id={s.id} className="flex scroll-mt-4 flex-col gap-2">
          {s.title && (
            <h2 className="font-display text-sm uppercase tracking-wide text-gold">{s.title}</h2>
          )}
          {s.blocks.map((b, i) => {
            if (b.type === "p") return <p key={i} className="text-sm text-muted">{b.text}</p>;
            if (b.type === "h") return <h3 key={i} className="font-display text-xs uppercase tracking-wide text-ink">{b.text}</h3>;
            return (
              <ul key={i} className="ml-4 flex list-disc flex-col gap-1 text-sm text-muted">
                {b.items.map((it, j) => <li key={j}>{it}</li>)}
              </ul>
            );
          })}
        </section>
      ))}
    </div>
  );
}
```
