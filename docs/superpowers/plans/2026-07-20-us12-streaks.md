# US-12 Système de Streaks — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Livrer la mécanique complète des streaks : série de victoires consécutives avec **budget de 2 non-victoires tolérées par semaine calendaire**, record persisté, compteur + état « en danger », titres de jalon.

**Architecture:** Tout le calcul est pur dans `lib/game-engine.ts` (`weekKey`, réécriture de `currentStreak` → objet, `titleForStreak`), testé Vitest. La série est **dérivée** de `day_results` ; `profiles.streak_days`/`best_streak`/`current_title` sont des caches rafraîchis à la clôture (`closeDay`/`backfill`). Affichage : compteur + danger au QG, flammes à la victoire (Crépuscule).

**Tech Stack:** Next.js 16 (App Router, server actions), Supabase (PostgreSQL), TypeScript strict, Vitest, Playwright (e2e série, prod).

## Global Constraints

- **TypeScript strict — pas de `any`.**
- **Logique pure dans `lib/game-engine.ts`** (testable sans UI).
- **Source unique = `day_results`.** `streak_days`/`best_streak`/`current_title` = caches, jamais la vérité.
- **Semaine = ISO (lundi→dimanche)**, calculée depuis le `day_date` (déjà une date locale — aucun fuseau à re-appliquer).
- **3 états UI** sur tout écran ; `prefers-reduced-motion` sur toute animation.
- **Réflexe non négociable** : toute US met à jour, dans le même commit/PR, (a) les **docs projet** pertinentes (`CLAUDE.md`, `CHANGELOG.md`, `SPECS_FONCTIONNELLES.md`) et (b) l'**issue GitHub** correspondante (bilan + `Ferme #NN` dans la PR). Tâches 8 & PR.
- **Budget jokers** : `WEEKLY_JOKER_BUDGET = 2` non-victoires tolérées / semaine ISO ; la 3ᵉ casse.
- **Changement de sémantique assumé** : un trou/non-victoire isolé n'est plus une rupture (c'est un joker toléré). Les tests existants de `currentStreak` sont réécrits en conséquence (Task 3).

```ts
export const WEEKLY_JOKER_BUDGET = 2;

export interface StreakState {
  length: number; // jours de victoire dans la chaîne
  jokersUsedThisWeek: number;
  atRisk: boolean;
}

// Semaine ISO 8601 (lundi→dimanche) à partir d'une date locale YYYY-MM-DD.
// Le day_date est déjà une date calendaire locale → aucun fuseau à ré-appliquer.
export function weekKey(dayDate: string): string {
  const date = new Date(`${dayDate}T00:00:00Z`);
  const dayNum = (date.getUTCDay() + 6) % 7; // lundi=0 … dimanche=6
  date.setUTCDate(date.getUTCDate() - dayNum + 3); // jeudi de la semaine ISO
  const isoYear = date.getUTCFullYear();
  const firstThursday = new Date(Date.UTC(isoYear, 0, 4));
  const firstDayNum = (firstThursday.getUTCDay() + 6) % 7;
  firstThursday.setUTCDate(firstThursday.getUTCDate() - firstDayNum + 3);
  const week = 1 + Math.round((date.getTime() - firstThursday.getTime()) / (7 * 86400000));
  return `${isoYear}-W${String(week).padStart(2, "0")}`;
}

// US-12 : série = victoires consécutives, avec un budget de 2 non-victoires
// (retreat/defeat/trou) tolérées par semaine ISO ; la 3e casse. Les jours
// tolérés bridgent la chaîne mais n'incrémentent pas `length`.
export function currentStreak(
  dayResults: Array<Pick<DayResult, "day_date" | "outcome">>,
  today: string,
): StreakState {
  if (dayResults.length === 0) {
    return { length: 0, jokersUsedThisWeek: 0, atRisk: false };
  }

  const outcomeByDate = new Map(dayResults.map((r) => [r.day_date, r.outcome]));
  const earliest = dayResults
    .map((r) => r.day_date)
    .reduce((min, d) => (d < min ? d : min));

  const jokersByWeek = new Map<string, number>();
  let length = 0;

  const cursor = new Date(`${today}T00:00:00Z`);
  // Aujourd'hui sans résultat : ni compté ni pénalisé, on part de la veille.
  if (!outcomeByDate.has(today)) {
    cursor.setUTCDate(cursor.getUTCDate() - 1);
  }

  while (true) {
    const ds = cursor.toISOString().slice(0, 10);
    if (ds < earliest) break; // plus d'historique

    if (outcomeByDate.get(ds) === "victory") {
      length += 1;
    } else {
      // non-victoire (retreat / defeat / trou) → consomme un joker de sa semaine
      const wk = weekKey(ds);
      const used = (jokersByWeek.get(wk) ?? 0) + 1;
      jokersByWeek.set(wk, used);
      if (used > WEEKLY_JOKER_BUDGET) break; // 3e non-victoire de la semaine → rupture
    }
    cursor.setUTCDate(cursor.getUTCDate() - 1);
  }

  const jokersUsedThisWeek = jokersByWeek.get(weekKey(today)) ?? 0;
  return {
    length,
    jokersUsedThisWeek,
    atRisk: jokersUsedThisWeek >= WEEKLY_JOKER_BUDGET,
  };
}

// US-12 : titre débloqué selon le record de série (provisoire — vocabulaire dojo,
// pourra fusionner avec d'éventuels titres de niveau plus tard).
export function titleForStreak(bestStreak: number): string | null {
  if (bestStreak >= 100) return "Sensei";
  if (bestStreak >= 30) return "Ceinture Noire";
  if (bestStreak >= 7) return "Ceinture Jaune";
  return null;
}
```

```ts
export async function seedStreak(
  rows: Array<{ day_date: string; outcome: "victory" | "retreat" | "defeat" }>,
): Promise<void> {
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const key = process.env.NEXT_PRIVATE_KEY_SUPABASE;
  const email = process.env.E2E_TEST_EMAIL;
  if (!url || !key || !email) throw new Error("Seed streak : env manquant.");
  const admin = createClient(url, key, { auth: { persistSession: false } });
  const { data: users } = await admin.auth.admin.listUsers();
  const uid = users?.users.find((u) => u.email === email)?.id;
  if (!uid) throw new Error("Seed streak : utilisateur test introuvable.");

  await admin.from("day_results").delete().eq("user_id", uid);
  await admin.from("profiles").update({ best_streak: 0, streak_days: 0 }).eq("id", uid);
  if (rows.length > 0) {
    await admin.from("day_results").upsert(
      rows.map((r) => ({
        user_id: uid,
        day_date: r.day_date,
        outcome: r.outcome,
        damage_dealt: 0,
        boss_max_hp: 0,
        tasks_completed: 0,
        tasks_abandoned: 0,
        xp_earned: 0,
        gold_earned: 0,
        closed_manually: true,
      })),
      { onConflict: "user_id,day_date" },
    );
  }
}
```
